# Playwright + Visual Regression Testing (VRT) Setup Guide

A comprehensive guide for setting up Playwright with Visual Regression Testing (VRT) in your project, including configuration for multiple browsers/devices, Drupal authentication support, and GitHub Actions integration.

---

## 1. Project Structure

Create the following directory structure:

```
/tests/playwright-vrt/
  ├── package.json
  ├── playwright.config.js
  ├── .env.example
  ├── .env (git ignored)
  ├── tests/
  │   ├── specs/
  │   │   └── visual_regression.spec.js
  │   ├── fixtures/
  │   │   ├── visual_regression_urls.js
  │   │   └── vrt-overrides.css
  │   └── utils/
  │       ├── vrt.js
  │       ├── auth.js (optional - for Drupal)
  │       ├── drupal-auth.js (optional - for Drupal)
  │       └── drush.js (optional - for Drupal)
  └── __snapshots__/ (generated by Playwright)
```

## 3. Playwright Configuration

Create `tests/playwright-vrt/playwright.config.js`:

```js
// playwright.config.js
require('dotenv').config();
const { devices } = require('@playwright/test');

/** @type {import('@playwright/test').PlaywrightTestConfig} */
const config = {
  testDir: './tests',
  timeout: 60 * 1000,
  expect: {
    timeout: 10000
  },
  reporter: [['list'], ['html', { outputFolder: 'playwright-report' }]],
  use: {
    headless: true,
    ignoreHTTPSErrors: true,
    screenshot: 'off',
    baseURL: process.env.BASE_URL,
  },

  // Projects = browser + device combinations
  // Each test runs in EVERY project (use --project flag to filter)
  projects: [
    {
      name: 'chrome-desktop',
      use: {
        ...devices['Desktop Chrome'],
      },
    },
    {
      name: 'firefox-desktop',
      use: {
        ...devices['Desktop Firefox'],
      },
    },
    {
      name: 'pixel-mobile',
      use: {
        ...devices['Pixel 7'],
      },
    },
    {
      name: 'iphone',
      use: {
        ...devices['iPhone 15 Pro'],
      },
    },
  ],
};

module.exports = config;
```

## 4. VRT Utilities

Create `tests/playwright-vrt/tests/utils/vrt.js`:

```js
/**
 * Visual Regression Testing (VRT) Utilities
 */

const { expect } = require('@playwright/test');

/**
 * Default screenshot options for visual regression tests.
 */
const DEFAULT_SCREENSHOT_OPTIONS = {
  fullPage: true,
  animations: 'disabled',
  maxDiffPixelRatio: 0.02
};

/**
 * Merge user-provided screenshot options with defaults.
 */
function mergeScreenshotOptions(overrides = {}) {
  return { ...DEFAULT_SCREENSHOT_OPTIONS, ...overrides };
}

/**
 * Normalize URL entry to object format.
 * Validates that required 'url' and 'name' properties exist.
 */
function normalizeUrlEntry(entry) {
  if (typeof entry === 'string') {
    throw new Error(
      `URL config must be an object with 'url' and 'name' properties. ` +
      `Got string: "${entry}". ` +
      `Use: { url: "${entry}", name: "..." }`
    );
  }

  if (!entry.url) {
    throw new Error(`URL config requires 'url' property.`);
  }

  if (!entry.name) {
    throw new Error(`URL config requires 'name' property for screenshot filenames.`);
  }

  return entry;
}

/**
 * Generate screenshot filename from name, project, and optional element name.
 */
function getScreenshotFilename(name, projectName, screenshotName) {
  const parts = [name];
  if (screenshotName) {
    parts.push(screenshotName);
  }
  return `${parts.join('-')}.png`;
}

/**
 * Capture screenshots for a page.
 * Supports both full-page screenshots and multiple element-specific screenshots.
 */
async function captureScreenshots(page, name, projectName, globalScreenshotOptions = {}, screenshotsConfig) {
  const baseOptions = mergeScreenshotOptions(globalScreenshotOptions);

  // If no screenshots config provided, capture full page
  if (!screenshotsConfig || screenshotsConfig.length === 0) {
    const filename = getScreenshotFilename(name, projectName);
    await expect(page).toHaveScreenshot(filename, baseOptions);
    return;
  }

  // Capture each configured screenshot
  for (const screenshotConfig of screenshotsConfig) {
    const { selector, name: elementName, options = {} } = screenshotConfig;

    if (!elementName) {
      throw new Error(`Screenshot config requires explicit 'name' property.`);
    }

    await page.locator(selector).waitFor({ state: 'attached', timeout: 5000 });
    
    const mergedOptions = { ...baseOptions, ...options };
    const filename = getScreenshotFilename(name, projectName, elementName);
    await expect(page.locator(selector)).toHaveScreenshot(filename, mergedOptions);
  }
}

module.exports = {
  DEFAULT_SCREENSHOT_OPTIONS,
  mergeScreenshotOptions,
  normalizeUrlEntry,
  getScreenshotFilename,
  captureScreenshots
};
```

## 6. Visual Regression Test Spec

Create `tests/playwright-vrt/tests/specs/visual_regression.spec.js`:

```js
const { test } = require('@playwright/test');
const config = require('../../playwright.config');
const visualRegressionUrls = require('../fixtures/visual_regression_urls.js');
const { normalizeUrlEntry, captureScreenshots } = require('../utils/vrt');

const BASE_URL = config.use.baseURL;

for (const entry of visualRegressionUrls) {
  const testConfig = normalizeUrlEntry(entry);
  const { url, name, screenshotOptions, screenshots, waitForSelector, projects } = testConfig;

  test.describe(`Visual regression for ${url}`, () => {
    test(`should match baseline for ${url}`, async ({ page }, testInfo) => {
      // Skip if this project is not in the allowed projects list
      if (projects && !projects.includes(testInfo.project.name)) {
        test.skip();
      }

      // Navigate to page
      await page.goto(`${BASE_URL}${url}`, { waitUntil: 'load', timeout: 30000 });
      
      // Optional: Apply CSS overrides
      // await page.addStyleTag({ path: './tests/fixtures/vrt-overrides.css' });
      
      await page.waitForLoadState('domcontentloaded');

      // Wait for custom selector if specified
      if (waitForSelector) {
        await page.locator(waitForSelector).waitFor({ state: 'attached' });
      }

      // Capture screenshots
      await captureScreenshots(page, name, testInfo.project.name, screenshotOptions, screenshots);
    });
  });
}
```

## 8. GitHub Actions CI Setup

### Option 1: Basic GitHub Actions Workflow

Create `.github/workflows/playwright-vrt.yml`:

```yaml
name: Playwright VRT

on:
  push:
    branches: [main]
  pull_request:

jobs:
  test:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: tests/playwright-vrt
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Install Playwright Browsers
        run: npx playwright install --with-deps
      
      - name: Run Playwright tests
        run: npm test
        env:
          BASE_URL: ${{ vars.BASE_URL || 'https://your-site.com' }}
      
      - name: Upload Playwright report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          path: tests/playwright-vrt/playwright-report
          retention-days: 30
```

### Option 2: Advanced Workflow with Manual Trigger & Baseline Refresh

Create `.github/workflows/playwright-tests.yml`:

```yaml
name: Playwright VRT

on:
  workflow_dispatch:
    inputs:
      mode:
        description: 'Test mode: baseline or test'
        required: true
        type: choice
        options:
          - test
          - baseline
        default: 'test'
      test_url:
        description: 'URL to test (for test mode)'
        required: false
  
  # Weekly baseline refresh
  schedule:
    - cron: '0 2 * * 1'

jobs:
  run-tests:
    if: github.event_name == 'workflow_dispatch' && inputs.mode == 'test'
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: tests/playwright-vrt
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Install Playwright Browsers
        run: npx playwright install --with-deps
      
      - name: Run Playwright tests
        run: npm test
        env:
          BASE_URL: ${{ inputs.test_url || vars.BASE_URL }}
      
      - name: Upload Playwright report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          path: tests/playwright-vrt/playwright-report
  
  refresh-baseline:
    if: github.event_name == 'schedule' || (github.event_name == 'workflow_dispatch' && inputs.mode == 'baseline')
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: tests/playwright-vrt
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Install Playwright Browsers
        run: npx playwright install --with-deps
      
      - name: Update baseline snapshots
        run: npm run test:baseline
        env:
          BASE_URL: ${{ vars.BASELINE_URL || vars.BASE_URL }}
      
      - name: Commit updated snapshots
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add tests/specs/**/*-snapshots/
          git commit -m "chore: update VRT baseline snapshots" || echo "No changes to commit"
          git push
```

## 9. Snapshot Management

### Directory Structure

Playwright organizes snapshots by project:

```
tests/specs/visual_regression.spec.js-snapshots/
  ├── chrome-desktop/
  │   ├── homepage.png
  │   └── about.png
  ├── firefox-desktop/
  │   ├── homepage.png
  │   └── about.png
  ├── pixel-mobile/
  │   └── homepage.png
  └── iphone/
      └── homepage.png
```

### Committing Snapshots

**Always commit your baseline snapshots to Git:**

```sh
git add tests/playwright-vrt/tests/specs/**/*-snapshots/
git commit -m "Add VRT baseline snapshots"
```

### Updating Baselines

When visual changes are intentional:

```sh
npm run test:baseline
git add tests/specs/**/*-snapshots/
git commit -m "Update VRT baselines for new design"
```

## 11. Troubleshooting

### Tests Fail Due to Minor Differences

Increase `maxDiffPixelRatio` in screenshot options:

```js
screenshotOptions: { maxDiffPixelRatio: 0.05 }
```

### Dynamic Content Causes Failures

Use CSS overrides to hide dynamic elements or disable animations:

```css
/* vrt-overrides.css */
.animated-element {
  animation: none !important;
}
```

### CI vs Local Differences

- Ensure consistent font rendering by installing system fonts in CI
- Use `headless: true` in both environments
- Consider using Docker for exact environment matching
